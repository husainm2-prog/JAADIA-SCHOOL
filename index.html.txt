        function renderStudentsSystem() {
            const selectEl = document.getElementById('active-student-select');
            const leaderboardEl = document.getElementById('leaderboard-list');
            const totalBadge = document.getElementById('total-students-badge');

            if (!studentsList.length) {

            const selectEl = document.getElementById('active-student-select');

            document.getElementById('active-student-grade').textContent = student.grade;

        function claimBehaviorPoint(reason, pointsToAdd) {
            const student = studentsList.find(s => s.id === activeStudentId);
            if (!student) {
                showToast('يرجى اختيار اسم الطالب أو استيراد قائمة الطلاب أولاً');
                return;
            }
            student.points += pointsToAdd;
            saveStudentsData();
            renderStudentsSystem();
            showToast(`أحسنت يا ${student.name}! تم إضافة +${pointsToAdd} نقاط (${reason})`);
        }

        }

        function handleStudentFileUpload(event) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                document.getElementById('bulk-students-textarea').value = e.target.result;
                showToast('تم تحميل الأسماء من الملف، اضغط "اعتماد واستيراد الأسماء"');
            };
            reader.readAsText(file);
        }

        function processBulkStudentImport() {
            const rawText = document.getElementById('bulk-students-textarea').value;
            const defaultGrade = document.getElementById('import-grade-level').value;
            const replaceExisting = document.getElementById('replace-existing-students').checked;

            const lines = rawText.split(/\r?\n/)
                .map(line => line.split(',')[0].replace(/["']/g, '').trim())
                .filter(name => name.length > 2);

            if (lines.length === 0) {
                showToast('يرجى إدخال اسم واحد على الأقل أو رفع ملف صالح');
                return;
            }

            const newEntries = lines.map((name, idx) => ({
                id: Date.now() + idx,
                name: name,
                grade: defaultGrade,
                points: 10
            }));

            if (replaceExisting) {
                studentsList = newEntries;
            } else {
                studentsList = [...studentsList, ...newEntries];
            }

            activeStudentId = newEntries[0].id;
            saveStudentsData();
            renderStudentsSystem();
            document.getElementById('bulk-students-textarea').value = '';
            closeImportStudentsModal();
            showToast(`تم استيراد ${newEntries.length} طالب للمرحلة الثانوية بنجاح!`);
        }

        function renderArticles() {
            const grid = document.getElementById('articles-grid');
            const noResults = document.getElementById('no-results');
            
            const filtered = articlesData.filter(article => {
                const matchCategory = currentCategory === 'all' || article.category === currentCategory;
                const searchLower = currentSearch.toLowerCase().trim();
                const matchSearch = !searchLower || 
                    article.title.toLowerCase().includes(searchLower) || 
                    article.excerpt.toLowerCase().includes(searchLower) ||
                    article.author.toLowerCase().includes(searchLower);
                return matchCategory && matchSearch;
            });

            if (filtered.length === 0) {
                grid.innerHTML = '';
                noResults.classList.remove('hidden');
                return;
            }

            noResults.classList.add('hidden');
            
            const colorMap = {
                blue: 'bg-blue-50 text-blue-600 dark:bg-blue-900/30 dark:text-blue-400',
                emerald: 'bg-emerald-50 text-emerald-600 dark:bg-emerald-900/30 dark:text-emerald-400',
                amber: 'bg-amber-50 text-amber-600 dark:bg-amber-900/30 dark:text-amber-400',
                purple: 'bg-purple-50 text-purple-600 dark:bg-purple-900/30 dark:text-purple-400'
            };

            grid.innerHTML = filtered.map(article => {
                const isBookmarked = bookmarkedIds.includes(article.id);
                return `
                <article class="bg-white dark:bg-slate-800 rounded-3xl p-6 border border-slate-100 dark:border-slate-700/80 shadow-sm hover:shadow-xl hover:-translate-y-1 transition-all duration-300 flex flex-col justify-between">
                    <div>
                        <div class="flex justify-between items-start gap-2 mb-4">
                            <span class="px-3 py-1 rounded-full text-xs font-bold ${colorMap[article.color] || colorMap.blue}">
                                <i class="fa-solid ${article.icon} ml-1"></i> ${article.categoryName}
                            </span>
                            <button onclick="toggleBookmark(${article.id}, event)" class="w-9 h-9 rounded-xl bg-slate-50 dark:bg-slate-700/60 hover:bg-amber-50 dark:hover:bg-slate-700 text-slate-400 hover:text-amber-500 transition flex items-center justify-center" title="حفظ المقال">
                                <i class="${isBookmarked ? 'fa-solid text-amber-500' : 'fa-regular'} fa-bookmark"></i>

        function filterByCategory(category) {
            currentCategory = category;
            document.querySelectorAll('.cat-btn').forEach(btn => {
                if (btn.dataset.cat === category) {
                    btn.className = 'cat-btn active px-5 py-2.5 rounded-xl font-bold text-sm bg-primary-600 text-white shadow-md shadow-primary-600/20 transition';
                } else {
                    btn.className = 'cat-btn px-5 py-2.5 rounded-xl font-bold text-sm bg-white dark:bg-slate-800 text-slate-600 dark:text-slate-300 border border-slate-200 dark:border-slate-700 hover:bg-slate-100 dark:hover:bg-slate-700 transition';
                }
            });
            renderArticles();
        }

        function resetFilters() {
            currentCategory = 'all';
            currentSearch = '';

        // ================= قراءة المقال والمحفوظات =================

        function openArticleModal(id) {
            const article = articlesData.find(a => a.id === id);
            if (!article) return;

            window.readArticlesSession = window.readArticlesSession || new Set();
            if (!window.readArticlesSession.has(id) && activeStudentId) {
                window.readArticlesSession.add(id);
                const student = studentsList.find(s => s.id === activeStudentId);
                if (student) {
                    student.points += 10;
                    saveStudentsData();
                    renderStudentsSystem();
                    showToast(`رائع يا ${student.name}! حصلت على +10 نقاط سلوك لقراءة المقال`);
                }
            }

            currentArticleId = id;
            document.getElementById('modal-category').textContent = article.categoryName;
            document.getElementById('modal-read-time').innerHTML = `<i class="fa-regular fa-clock ml-1"></i> ${article.readTime}`;
            document.getElementById('modal-title').textContent = article.title;
            document.getElementById('modal-author').textContent = article.author;
            document.getElementById('modal-author-role').textContent = article.authorRole;
            document.getElementById('modal-author-avatar').textContent = article.avatar;

            const bodyContainer = document.getElementById('modal-body');
            bodyContainer.style.fontSize = `${currentFontSize}px`;
            bodyContainer.innerHTML = article.content.map(paragraph => `<p>${paragraph}</p>`).join('');

            updateModalBookmarkBtn();
            document.getElementById('article-modal').classList.remove('hidden');
            document.body.style.overflow = 'hidden';
        }

        function closeArticleModal() {
            document.getElementById('article-modal').classList.add('hidden');
            document.body.style.overflow = 'auto';
        }

        function changeFontSize(delta) {
            currentFontSize = Math.min(22, Math.max(14, currentFontSize + delta * 2));
            document.getElementById('modal-body').style.fontSize = `${currentFontSize}px`;
        }

        function toggleBookmark(id, event) {
            if (event) event.stopPropagation();
            const index = bookmarkedIds.indexOf(id);
            if (index === -1) {
                bookmarkedIds.push(id);
                showToast('تم حفظ المقال في قائمة القراءة');
            } else {
                bookmarkedIds.splice(index, 1);
                showToast('تمت إزالة المقال من المحفوظات');
            }
            localStorage.setItem('guidance_bookmarks', JSON.stringify(bookmarkedIds));
            updateBookmarkBadge();
            renderArticles();
            if (currentArticleId === id) updateModalBookmarkBtn();
        }

        function toggleCurrentBookmark() {
            if (currentArticleId) toggleBookmark(currentArticleId);
        }

        function updateModalBookmarkBtn() {
            const btn = document.getElementById('modal-bookmark-btn');
            const isSaved = bookmarkedIds.includes(currentArticleId);
            if (isSaved) {
                btn.innerHTML = `<i class="fa-solid fa-bookmark text-amber-500"></i> <span>محفوظ</span>`;
            } else {
                btn.innerHTML = `<i class="fa-regular fa-bookmark"></i> <span>حفظ المقال</span>`;
            }
        }

        function updateBookmarkBadge() {
            const badge = document.getElementById('bookmark-count');
            if (bookmarkedIds.length > 0) {
                badge.textContent = bookmarkedIds.length;
                badge.classList.remove('hidden');
            } else {
                badge.classList.add('hidden');
            }
        }

        function openBookmarksModal() {
            const list = document.getElementById('bookmarks-list');
            const savedArticles = articlesData.filter(a => bookmarkedIds.includes(a.id));
            
            if (savedArticles.length === 0) {
                list.innerHTML = `
                    <div class="text-center py-8 text-slate-400">
                        <i class="fa-regular fa-bookmark text-4xl mb-2"></i>
                        <p class="text-sm">لا توجد مقالات محفوظة حالياً.</p>
                    </div>
                `;
            } else {
                list.innerHTML = savedArticles.map(a => `
                    <div class="p-3.5 rounded-2xl bg-slate-50 dark:bg-slate-700/50 flex justify-between items-center gap-3">
                        <div class="cursor-pointer" onclick="closeBookmarksModal(); openArticleModal(${a.id})">
                            <span class="text-[11px] font-bold text-primary-600 dark:text-primary-400">${a.categoryName}</span>
                            <h4 class="font-bold text-sm text-slate-800 dark:text-white line-clamp-1">${a.title}</h4>
                        </div>
                        <button onclick="toggleBookmark(${a.id}); openBookmarksModal()" class="text-rose-500 hover:bg-rose-50 dark:hover:bg-slate-600 p-2 rounded-lg text-xs" title="حذف">
                            <i class="fa-solid fa-trash-can"></i>
                        </button>
                    </div>
                `).join('');
            }
            document.getElementById('bookmarks-modal').classList.remove('hidden');
        }

        function closeBookmarksModal() {
            document.getElementById('bookmarks-modal').classList.add('hidden');
        }

        function copyArticleLink() {
            const tempInput = document.createElement('input');
            tempInput.value = window.location.href;
            document.body.appendChild(tempInput);
            tempInput.select();
            document.execCommand('copy');
            document.body.removeChild(tempInput);
            showToast('تم نسخ رابط المقال لمشاركته');
        }

        function evaluateStudyHabits() {
            const score = parseInt(document.getElementById('q1').value) + 
                          parseInt(document.getElementById('q2').value) + 
                          parseInt(document.getElementById('q3').value);
            
            const resultBox = document.getElementById('quiz-result');
            let title = '';
            let advice = '';

            if (score >= 8) {
                title = '🌟 مستوى متميز! (عادات دراسية صحية)';
                advice = 'أنت تمتلك تنظيماً رائعاً لوقتك ونومك. استمر على هذا النهج، وننصحك بقراءة مقال "مسارات المرحلة الثانوية" للتخطيط المبكر لتخصصك الجامعي.';
            } else if (score >= 5) {
                title = '👍 مستوى جيد (يحتاج إلى بعض الضبط)';
                advice = 'لديك أساس جيد، لكن تقليل المشتتات الرقمية وتثبيت ساعات النوم سيرفعان من تركيزك بشكل ملحوظ. راجع ملف العدد الرئيسي لخطوات عملية.';
            } else {
                title = '💡 فرصة رائعة للتحسين!';
                advice = 'يبدو أنك تواجه تشتتاً وضغطاً في تنظيم وقتك. ابدأ اليوم بخطوة بسيطة: ضع الهاتف في غرفة أخرى لمدة 30 دقيقة فقط أثناء المذاكرة، وتواصل مع الموجه الطلابي لمساعدتك في بناء جدول مريح.';
            }

            const student = studentsList.find(s => s.id === activeStudentId);
            if (student) {
                student.points += 5;
                saveStudentsData();
                renderStudentsSystem();
                advice += ` (تمت إضافة +5 نقاط سلوك لرصيد الطالب: ${student.name})`;
            }

            resultBox.innerHTML = `<h4 class="font-bold text-primary-800 dark:text-primary-300 mb-1">${title}</h4><p class="text-slate-700 dark:text-slate-300 text-xs leading-relaxed">${advice}</p>`;
            resultBox.classList.remove('hidden');
        }

        function toggleDarkMode() {
            const html = document.documentElement;
            const icon = document.getElementById('theme-icon');
            html.classList.toggle('dark');
            
            if (html.classList.contains('dark')) {
                icon.classList.replace('fa-moon', 'fa-sun');
                localStorage.setItem('guidance_theme', 'dark');
            } else {
                icon.classList.replace('fa-sun', 'fa-moon');
                localStorage.setItem('guidance_theme', 'light');
            }
        }

            toast.classList.remove('hidden');

            window.toastTimeout = setTimeout(() => {
                toast.classList.add('hidden');
